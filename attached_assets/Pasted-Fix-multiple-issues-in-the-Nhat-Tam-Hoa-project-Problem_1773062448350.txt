STEP 5 GENERATE CERTIFICATE
================================================

Example certificate generator:

import PDFDocument from "pdfkit"
import QRCode from "qrcode"
import fs from "fs"


export async function generateCertificate(order){

const certCode = order.certificate_id

const verifyUrl =
"https://nhattamhoa-tranminhhanhqn.replit.app/certificate/" + certCode


const doc = new PDFDocument({
size:"A4",
margin:50
})


const filePath =
"./public/certificates/certificate-" + certCode + ".pdf"


doc.pipe(fs.createWriteStream(filePath))


doc.font("fonts/DejaVuSans.ttf")


doc.fontSize(30)
.text("NHẤT TÂM HOA",{align:"center"})


doc.moveDown()

doc.fontSize(14)
.text("Flower Intention Certificate",{align:"center"})


doc.moveDown(2)


doc.text("Sender: " + order.sender_name)
doc.text("Receiver: " + order.receiver_name)


doc.moveDown()

doc.text("Message:")
doc.text(order.message)



================================================
STEP 6 ADD QR CODE
================================================

const qrImage =
await QRCode.toDataURL(verifyUrl)


doc.image(
Buffer.from(qrImage.split(",")[1],"base64"),
420,
520,
{width:120}
)


doc.moveDown()

doc.text("Verify certificate:")
doc.text(verifyUrl)


doc.end()

return filePath

}
