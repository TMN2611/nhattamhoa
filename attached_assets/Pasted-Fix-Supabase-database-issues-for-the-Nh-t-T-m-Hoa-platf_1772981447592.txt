2 FIX CREATE ORDER API
================================================

File:

/api/create-order.js

Function must:

1 generate certificate code
2 generate blockchain hash
3 insert order
4 create certificate record


Example logic:

import { supabase } from "@/lib/supabaseClient";
import crypto from "crypto";

export async function POST(req) {

const body = await req.json();

const certificateCode =
"NTT-" + Date.now();

const hash = crypto
.createHash("sha256")
.update(
body.sender +
body.receiver +
body.message +
Date.now()
)
.digest("hex");


const { data, error } = await supabase
.from("orders")
.insert([
{
sender_name: body.sender,
receiver_name: body.receiver,
message: body.message,
ritual_type: body.ritual,
offering: body.offering,
certificate_id: certificateCode,
blockchain_hash: hash,
public_vow: body.publicVow
}
])
.select()
.single();


if (error) {
console.error("Order insert error:", error);
return Response.json({error});
}


await supabase.from("certificates").insert([
{
certificate_code: certificateCode,
order_id: data.id,
blockchain_hash: hash
}
]);


return Response.json({
success:true,
certificate_code: certificateCode
});
}
